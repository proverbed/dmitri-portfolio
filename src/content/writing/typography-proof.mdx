## Code

```ts
export function readingTime(source: string): number {
  const fences = source.match(/```[\s\S]*?```/g) ?? [];
  const prose = source.replace(/```[\s\S]*?```/g, " ");
  const words = prose.split(/\s+/).filter(Boolean).length;
  return Math.max(1, Math.round(words / 220));
}
```
